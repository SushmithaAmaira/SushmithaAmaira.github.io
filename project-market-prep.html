PDF Generation Script

generate_pdf.pyPython
from reportlab.platypus import SimpleDocTemplate, Paragraph, Table
from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle
from reportlab.lib.pagesizes import A4
from reportlab.lib.units import mm

# Create the PDF document
doc = SimpleDocTemplate(
    "Kite_Connect_Full_Study_Notes.pdf",
    pagesize=A4,
    rightMargin=20*mm, leftMargin=20*mm
)

styles = getSampleStyleSheet()
story = []

# Section: What is Kite Connect API
story.append(Paragraph("What is Zerodha Kite Connect API?", styles['Heading1']))
story.append(Paragraph(
    "Kite Connect is Zerodha's set of REST APIs that lets developers build "
    "trading platforms and investment apps on top of Zerodha's infrastructure.",
    styles['Normal']
))

# Table of all 41 KiteConnect functions
data = [
    ['Function', 'Description'],
    ['login_url()', 'Returns the login URL for OAuth'],
    ['generate_session()', 'Generates an access token'],
    ['place_order()', 'Places a new order'],
    # ... 38 more functions
]

story.append(Table(data))
doc.build(story)
print("PDF created successfully!")
